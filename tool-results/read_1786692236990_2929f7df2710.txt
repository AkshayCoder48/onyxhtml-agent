   523→// Build a short human hint for a tool call, used in tool_call/tool_result events.
   524→export function getToolDetail(name: string, args: Record<string, unknown>): string {
   525→  try {
   526→    switch (name) {
   527→      case "list_files":
   528→        return "all files";
   529→      case "read_file":
   530→      case "create_file":
   531→      case "write_file":
   532→      case "edit_file":
   533→      case "delete_file":
   534→      case "create_folder":
   535→        return String(args.path ?? "");
   536→      case "rename_file":
   537→      case "move_file":
   538→        return `${args.from ?? ""} → ${args.to ?? ""}`;
   539→      case "replace_content":
   540→        return String(args.path ?? "");
   541→      case "search_files":
   542→        return String(args.query ?? "");
   543→      case "open_page":
   544→        return typeof args.url === "string" ? args.url : "preview";
   545→      case "reload_page":
   546→        return "reload";
   547→      case "click":
   548→      case "hover":
   549→      case "get_dom":
   550→      case "get_element":
   551→      case "inspect_element":
   552→        return String(args.selector ?? "");
   553→      case "type":
   554→        return String(args.selector ?? "");
   555→      case "press_key":
   556→        return String(args.key ?? "");
   557→      case "scroll":
   558→        return args.selector ? String(args.selector) : "window";
   559→      case "select":
   560→        return String(args.selector ?? "");
   561→      case "wait":
   562→        return `${args.ms ?? 0}ms`;
   563→      case "take_screenshot":
   564→        return "screenshot";
   565→      case "run_javascript":
   566→        return "JS";
   567→      case "run_test":
   568→        return typeof args.name === "string" ? args.name : "test";
   569→      case "get_console_logs":
   570→        return "console";
   571→      case "get_page_errors":
   572→        return "errors";
   573→      case "get_network_errors":
   574→        return "network";
   575→      case "check_page":
   576→        return "page";
   577→      case "check_console":
   578→        return "console";
   579→      case "check_links":
   580→        return "links";
   581→      default:
   582→        return "";
   583→    }
   584→  } catch {
   585→    return "";
   586→  }
   587→}

   590→  return (TOOL_LABELS as Record<string, string>)[name] ?? name;

   593→// ---------- Server-side file tool executor ----------
   594→
   595→export async function executeFileTool(
   596→  name: string,
   597→  args: Record<string, unknown>,
   598→  workspaceId: string
   599→): Promise<unknown> {
   600→  switch (name) {
   601→    case "list_files": {
   602→      const files = await db.file.findMany({
   603→        where: { workspaceId },
   604→        select: { path: true },
   605→        orderBy: { path: "asc" },
   606→      });
   607→      return { paths: files.map((f) => f.path) };
   608→    }
   609→    case "read_file": {
   610→      const path = safePath(String(args.path ?? ""));
   611→      if (!path) throw new Error("Invalid path");
   612→      const file = await db.file.findUnique({
   613→        where: { workspaceId_path: { workspaceId, path } },
   614→      });
   615→      if (!file) throw new Error(`File not found: ${path}`);
   616→      const lines = file.content.split("\n").length;
   617→      return { path: file.path, content: file.content, lines };
   618→    }
   619→    case "create_file":
   620→    case "write_file": {
   621→      const path = safePath(String(args.path ?? ""));
   622→      if (!path) throw new Error("Invalid path");
   623→      const content = String(args.content ?? "");
   624→      const isBinary = false;
   625→      const file = await db.file.upsert({
   626→        where: { workspaceId_path: { workspaceId, path } },
   627→        update: { content, isBinary },
   628→        create: { workspaceId, path, content, isBinary },
   629→      });
   630→      return { path: file.path };
   631→    }
   632→    case "edit_file": {
   633→      const path = safePath(String(args.path ?? ""));
   634→      if (!path) throw new Error("Invalid path");
   635→      const oldContent = String(args.oldContent ?? "");
   636→      const newContent = String(args.newContent ?? "");
   637→      const file = await db.file.findUnique({
   638→        where: { workspaceId_path: { workspaceId, path } },
   639→      });
   640→      if (!file) throw new Error(`File not found: ${path}`);
   641→      const idx = file.content.indexOf(oldContent);
   642→      if (idx === -1) {
   643→        throw new Error(
   644→          `oldContent not found in ${path}. Make sure it matches exactly (including whitespace).`
   645→        );
   646→      }
   647→      const next =
   648→        file.content.slice(0, idx) + newContent + file.content.slice(idx + oldContent.length);
   649→      await db.file.update({
   650→        where: { id: file.id },
   651→        data: { content: next },
   652→      });
   653→      return { path, replaced: 1 };
   654→    }
   655→    case "delete_file": {
   656→      const path = safePath(String(args.path ?? ""));
   657→      if (!path) throw new Error("Invalid path");
   658→      const prefix = path.endsWith("/") ? path : path + "/";
   659→      await db.file.deleteMany({
   660→        where: {
   661→          workspaceId,
   662→          OR: [{ path }, { path: { startsWith: prefix } }],
   663→        },
   664→      });
   665→      return { path };
   666→    }
   667→    case "rename_file":
   668→    case "move_file": {
   669→      const from = safePath(String(args.from ?? ""));
   670→      const to = safePath(String(args.to ?? ""));
   671→      if (!from || !to) throw new Error("Invalid path");
   672→      if (from === to) throw new Error("Source and destination are the same");
   673→      const existing = await db.file.findUnique({
   674→        where: { workspaceId_path: { workspaceId, path: from } },
   675→      });
   676→      const children = await db.file.findMany({
   677→        where: { workspaceId, path: { startsWith: from + "/" } },
   678→      });
   679→      if (!existing && children.length === 0) {
   680→        throw new Error(`File or folder not found: ${from}`);
   681→      }
   682→      const destExists = await db.file.findUnique({
   683→        where: { workspaceId_path: { workspaceId, path: to } },
   684→      });
   685→      if (destExists) throw new Error(`Destination already exists: ${to}`);
   686→      // If `from` is a folder, also reject if `to` is inside `from` (would create a cycle).
   687→      if (children.length > 0 && (to + "/").startsWith(from + "/")) {
   688→        throw new Error("Cannot move a folder into itself");
   689→      }
   690→
   691→      await db.$transaction(async (tx) => {
   692→        if (existing) {
   693→          await tx.file.update({
   694→            where: { id: existing.id },
   695→            data: { path: to },
   696→          });
   697→        }
   698→        for (const child of children) {
   699→          const newChildPath = to + child.path.slice(from.length);
   700→          await tx.file.update({
   701→            where: { id: child.id },
   702→            data: { path: newChildPath },
   703→          });
   704→        }
   705→      });
   706→      return { from, to };
   707→    }
   708→    case "replace_content": {
   709→      const path = safePath(String(args.path ?? ""));
   710→      if (!path) throw new Error("Invalid path");
   711→      const find = String(args.find ?? "");
   712→      if (!find) throw new Error("`find` must not be empty");
   713→      const replace = String(args.replace ?? "");
   714→      const all = args.all !== false; // default true
   715→      const file = await db.file.findUnique({
   716→        where: { workspaceId_path: { workspaceId, path } },
   717→      });
   718→      if (!file) throw new Error(`File not found: ${path}`);
   719→      const content = file.content;
   720→      let count = 0;
   721→      let next: string;
   722→      if (all) {
   723→        if (find === replace) {
   724→          // No-op but still count occurrences for the response.
   725→          count = content.split(find).length - 1;
   726→          next = content;
   727→        } else {
   728→          // Use split/join to replace all occurrences without regex escaping concerns.
   729→          const parts = content.split(find);
   730→          count = parts.length - 1;
   731→          next = parts.join(replace);
   732→        }
   733→      } else {
   734→        const idx = content.indexOf(find);
   735→        if (idx === -1) {
   736→          throw new Error(
   737→            `Text not found in ${path}. Make sure \`find\` matches exactly (including whitespace).`
   738→          );
   739→        }
   740→        next = content.slice(0, idx) + replace + content.slice(idx + find.length);
   741→        count = 1;
   742→      }
   743→      if (count === 0) {
   744→        throw new Error(
   745→          `Text not found in ${path}. Make sure \`find\` matches exactly (including whitespace).`
   746→        );
   747→      }
   748→      await db.file.update({
   749→        where: { id: file.id },
   750→        data: { content: next },
   751→      });
   752→      return { path, replaced: count };
   753→    }
   754→    case "create_folder": {
   755→      const folder = safePath(String(args.path ?? ""));
   756→      if (!folder) throw new Error("Invalid path");
   757→      const keepPath = folder.endsWith("/") ? folder + ".gitkeep" : folder + "/.gitkeep";
   758→      await db.file.upsert({
   759→        where: { workspaceId_path: { workspaceId, path: keepPath } },
   760→        update: {},
   761→        create: { workspaceId, path: keepPath, content: "", isBinary: false },
   762→      });
   763→      return { path: folder };
   764→    }
   765→    case "search_files": {
   766→      const query = String(args.query ?? "").toLowerCase();
   767→      if (!query) return { matches: [] };
   768→      const files = await db.file.findMany({
   769→        where: { workspaceId },
   770→        select: { path: true, content: true },
   771→      });
   772→      const matches: { path: string; line: number; text: string }[] = [];
   773→      outer: for (const f of files) {
   774→        const lines = f.content.split("\n");
   775→        for (let i = 0; i < lines.length; i++) {
   776→          if (lines[i].toLowerCase().includes(query)) {
   777→            matches.push({
   778→              path: f.path,
   779→              line: i + 1,
   780→              text: lines[i].trim().slice(0, 200),
   781→            });
   782→            if (matches.length >= 50) break outer;
   783→          }
   784→        }
   785→      }
   786→      return { matches };
   787→    }
   788→    default:
   789→      throw new Error(`Unknown file tool: ${name}`);
   790→  }
