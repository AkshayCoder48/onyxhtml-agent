   552→// Build a short human hint for a tool call, used in tool_call/tool_result events.
   553→export function getToolDetail(name: string, args: Record<string, unknown>): string {
   554→  try {
   555→    switch (name) {
   556→      case "list_files":
   557→        return "all files";
   558→      case "read_file":
   559→      case "create_file":
   560→      case "write_file":
   561→      case "edit_file":
   562→      case "delete_file":
   563→      case "create_folder":
   564→        return String(args.path ?? "");
   565→      case "rename_file":
   566→      case "move_file":
   567→        return `${args.from ?? ""} → ${args.to ?? ""}`;
   568→      case "replace_content":
   569→        return String(args.path ?? "");
   570→      case "search_files":
   571→        return String(args.query ?? "");
   572→      case "open_page":
   573→        return typeof args.url === "string" ? args.url : "preview";
   574→      case "reload_page":
   575→        return "reload";
   576→      case "click":
   577→      case "hover":
   578→      case "get_dom":
   579→      case "get_element":
   580→      case "inspect_element":
   581→        return String(args.selector ?? "");
   582→      case "type":
   583→        return String(args.selector ?? "");
   584→      case "press_key":
   585→        return String(args.key ?? "");
   586→      case "scroll":
   587→        return args.selector ? String(args.selector) : "window";
   588→      case "select":
   589→        return String(args.selector ?? "");
   590→      case "wait":
   591→        return `${args.ms ?? 0}ms`;
   592→      case "take_screenshot":
   593→        return "screenshot";
   594→      case "run_javascript":
   595→        return "JS";
   596→      case "run_test":
   597→        return typeof args.name === "string" ? args.name : "test";
   598→      case "terminal_exec":
   599→        // Show the first line of the code as the detail (truncated).
   600→        return String(args.code ?? "")
   601→          .split("\n")[0]
   602→          .slice(0, 60);
   603→      case "terminal_reset":
   604→        return "clear";
   605→      case "get_console_logs":
   606→        return "console";
   607→      case "get_page_errors":
   608→        return "errors";
   609→      case "get_network_errors":
   610→        return "network";
   611→      case "check_page":
   612→        return "page";
   613→      case "check_console":
   614→        return "console";
   615→      case "check_links":
   616→        return "links";
   617→      default:
   618→        return "";
   619→    }
   620→  } catch {
   621→    return "";
   622→  }
   623→}
   624→
   625→export function getToolLabel(name: string): string {
   626→  return (TOOL_LABELS as Record<string, string>)[name] ?? name;
   627→}

   629→// ---------- Server-side file tool executor ----------
   630→
   631→export async function executeFileTool(
   632→  name: string,
   633→  args: Record<string, unknown>,
   634→  workspaceId: string
   635→): Promise<unknown> {
   636→  switch (name) {
   637→    case "list_files": {
   638→      const files = await db.file.findMany({
   639→        where: { workspaceId },
   640→        select: { path: true },
   641→        orderBy: { path: "asc" },
   642→      });
   643→      return { paths: files.map((f) => f.path) };
   644→    }
   645→    case "read_file": {
   646→      const path = safePath(String(args.path ?? ""));
   647→      if (!path) throw new Error("Invalid path");
   648→      const file = await db.file.findUnique({
   649→        where: { workspaceId_path: { workspaceId, path } },
   650→      });
   651→      if (!file) throw new Error(`File not found: ${path}`);
   652→      const lines = file.content.split("\n").length;
   653→      return { path: file.path, content: file.content, lines };
   654→    }
   655→    case "create_file":
   656→    case "write_file": {
   657→      const path = safePath(String(args.path ?? ""));
   658→      if (!path) throw new Error("Invalid path");
   659→      const content = String(args.content ?? "");
   660→      const isBinary = false;
   661→      const file = await db.file.upsert({
   662→        where: { workspaceId_path: { workspaceId, path } },
   663→        update: { content, isBinary },
   664→        create: { workspaceId, path, content, isBinary },
   665→      });
   666→      return { path: file.path };
   667→    }
   668→    case "edit_file": {
   669→      const path = safePath(String(args.path ?? ""));
   670→      if (!path) throw new Error("Invalid path");
   671→      const oldContent = String(args.oldContent ?? "");
   672→      const newContent = String(args.newContent ?? "");
   673→      const file = await db.file.findUnique({
   674→        where: { workspaceId_path: { workspaceId, path } },
   675→      });
   676→      if (!file) throw new Error(`File not found: ${path}`);
   677→      const idx = file.content.indexOf(oldContent);
   678→      if (idx === -1) {
   679→        throw new Error(
   680→          `oldContent not found in ${path}. Make sure it matches exactly (including whitespace).`
   681→        );
   682→      }
   683→      const next =
   684→        file.content.slice(0, idx) + newContent + file.content.slice(idx + oldContent.length);
   685→      await db.file.update({
   686→        where: { id: file.id },
   687→        data: { content: next },
   688→      });
   689→      return { path, replaced: 1 };
   690→    }
   691→    case "delete_file": {
   692→      const path = safePath(String(args.path ?? ""));
   693→      if (!path) throw new Error("Invalid path");
   694→      const prefix = path.endsWith("/") ? path : path + "/";
   695→      await db.file.deleteMany({
   696→        where: {
   697→          workspaceId,
   698→          OR: [{ path }, { path: { startsWith: prefix } }],
   699→        },
   700→      });
   701→      return { path };
   702→    }
   703→    case "rename_file":
   704→    case "move_file": {
   705→      const from = safePath(String(args.from ?? ""));
   706→      const to = safePath(String(args.to ?? ""));
   707→      if (!from || !to) throw new Error("Invalid path");
   708→      if (from === to) throw new Error("Source and destination are the same");
   709→      const existing = await db.file.findUnique({
   710→        where: { workspaceId_path: { workspaceId, path: from } },
   711→      });
   712→      const children = await db.file.findMany({
   713→        where: { workspaceId, path: { startsWith: from + "/" } },
   714→      });
   715→      if (!existing && children.length === 0) {
   716→        throw new Error(`File or folder not found: ${from}`);
   717→      }
   718→      const destExists = await db.file.findUnique({
   719→        where: { workspaceId_path: { workspaceId, path: to } },
   720→      });
   721→      if (destExists) throw new Error(`Destination already exists: ${to}`);
   722→      // If `from` is a folder, also reject if `to` is inside `from` (would create a cycle).
   723→      if (children.length > 0 && (to + "/").startsWith(from + "/")) {
   724→        throw new Error("Cannot move a folder into itself");
   725→      }
   726→
   727→      await db.$transaction(async (tx) => {
   728→        if (existing) {
   729→          await tx.file.update({
   730→            where: { id: existing.id },
   731→            data: { path: to },
   732→          });
   733→        }
   734→        for (const child of children) {
   735→          const newChildPath = to + child.path.slice(from.length);
   736→          await tx.file.update({
   737→            where: { id: child.id },
   738→            data: { path: newChildPath },
   739→          });
   740→        }
   741→      });
   742→      return { from, to };
   743→    }
   744→    case "replace_content": {
   745→      const path = safePath(String(args.path ?? ""));
   746→      if (!path) throw new Error("Invalid path");
   747→      const find = String(args.find ?? "");
   748→      if (!find) throw new Error("`find` must not be empty");
   749→      const replace = String(args.replace ?? "");
   750→      const all = args.all !== false; // default true
   751→      const file = await db.file.findUnique({
   752→        where: { workspaceId_path: { workspaceId, path } },
   753→      });
   754→      if (!file) throw new Error(`File not found: ${path}`);
   755→      const content = file.content;
   756→      let count = 0;
   757→      let next: string;
   758→      if (all) {
   759→        if (find === replace) {
   760→          // No-op but still count occurrences for the response.
   761→          count = content.split(find).length - 1;
   762→          next = content;
   763→        } else {
   764→          // Use split/join to replace all occurrences without regex escaping concerns.
   765→          const parts = content.split(find);
   766→          count = parts.length - 1;
   767→          next = parts.join(replace);
   768→        }
   769→      } else {
   770→        const idx = content.indexOf(find);
   771→        if (idx === -1) {
   772→          throw new Error(
   773→            `Text not found in ${path}. Make sure \`find\` matches exactly (including whitespace).`
   774→          );
   775→        }
   776→        next = content.slice(0, idx) + replace + content.slice(idx + find.length);
   777→        count = 1;
   778→      }
   779→      if (count === 0) {
   780→        throw new Error(
   781→          `Text not found in ${path}. Make sure \`find\` matches exactly (including whitespace).`
   782→        );
   783→      }
   784→      await db.file.update({
   785→        where: { id: file.id },
   786→        data: { content: next },
   787→      });
   788→      return { path, replaced: count };
   789→    }
   790→    case "create_folder": {
   791→      const folder = safePath(String(args.path ?? ""));
   792→      if (!folder) throw new Error("Invalid path");
   793→      const keepPath = folder.endsWith("/") ? folder + ".gitkeep" : folder + "/.gitkeep";
   794→      await db.file.upsert({
   795→        where: { workspaceId_path: { workspaceId, path: keepPath } },
   796→        update: {},
   797→        create: { workspaceId, path: keepPath, content: "", isBinary: false },
   798→      });
   799→      return { path: folder };
   800→    }
   801→    case "search_files": {
   802→      const query = String(args.query ?? "").toLowerCase();
   803→      if (!query) return { matches: [] };
   804→      const files = await db.file.findMany({
   805→        where: { workspaceId },
   806→        select: { path: true, content: true },
   807→      });
   808→      const matches: { path: string; line: number; text: string }[] = [];
   809→      outer: for (const f of files) {
   810→        const lines = f.content.split("\n");
   811→        for (let i = 0; i < lines.length; i++) {
   812→          if (lines[i].toLowerCase().includes(query)) {
   813→            matches.push({
   814→              path: f.path,
   815→              line: i + 1,
   816→              text: lines[i].trim().slice(0, 200),
   817→            });
   818→            if (matches.length >= 50) break outer;
   819→          }
   820→        }
   821→      }
   822→      return { matches };
   823→    }
   824→    default:
   825→      throw new Error(`Unknown file tool: ${name}`);
   826→  }
