   300→   300→  const dispatcherRef = React.useRef(new EventDispatcher(handlers));
   301→   301→  React.useEffect(() => {
   302→   302→    // Update the dispatcher's handlers when the memoized handlers change
   303→   303→    // (they're stable across renders due to the dependency array above, but
   304→   304→    // we keep this effect to be safe).

   307→   307→
   308→   308→  // ---------- Stream consumer ----------
   309→   309→  //
   310→   310→  // Consumes a raw ReadableStream via the StreamEngine, dispatches each parsed
   311→   311→  // StreamEvent to the stores IMMEDIATELY, and collects any browser tool
   312→   312→  // callIds that need client-side execution. Returns the list of pending
   313→   313→  // browser tool callIds when the stream ends.
   314→   314→  const consumeStream = React.useCallback(
   315→   315→    async (
   316→   316→      stream: ReadableStream<Uint8Array>,
   317→   317→      assistantId: string,
   318→   318→      signal: AbortSignal
   319→   319→    ): Promise<{ browserPending: string[] }> => {
   320→   320→      const browserPending: string[] = [];
   321→   321→      const dispatcher = dispatcherRef.current;
   322→   322→
   323→   323→      // Watchdog: if no event arrives for 5 minutes, abort so the spinner
   324→   324→      // can't spin forever (PRD §33 interrupted streams). We use the outer
   325→   325→      // abortRef (set by sendMessage) to cancel the in-flight fetch.
   326→   326→      let watchdogTimer: ReturnType<typeof setTimeout> | null = null;
   327→   327→      let timedOut = false;
   328→   328→      const WATCHDOG_MS = 5 * 60 * 1000;
   329→   329→      const resetWatchdog = () => {
   330→   330→        if (watchdogTimer) clearTimeout(watchdogTimer);
   331→   331→        watchdogTimer = setTimeout(() => {
   332→   332→          timedOut = true;
   333→   333→          addErrorSegment(
   334→   334→            assistantId,
   335→   335→            "The AI stream has been silent for 5 minutes and was aborted. Please try again."
   336→   336→          );
   337→   337→          toast.error("AI stream timed out", {
   338→   338→            description: "No events received for 5 minutes.",
   339→   339→          });
   340→   340→          try {
   341→   341→            abortRef.current?.abort();
   342→   342→          } catch {
   343→   343→            // noop
   344→   344→          }
   345→   345→        }, WATCHDOG_MS);
   346→   346→      };
   347→   347→      const clearWatchdog = () => {
   348→   348→        if (watchdogTimer) {
   349→   349→          clearTimeout(watchdogTimer);
   350→   350→          watchdogTimer = null;
   351→   351→        }
   352→   352→      };
   353→   353→      resetWatchdog();
   354→   354→
   355→   355→      // CRITICAL: The client creates the assistant message with a local ID
   356→   356→      // (e.g. "a_abc123"), but the server sends stream events with the DB
   357→   357→      // message's ID (e.g. "cmss..."). If we don't reconcile these IDs,
   358→   358→      // every appendTextDelta / upsertToolCallSegment call will fail to find
   359→   359→      // the message and silently drop the update — causing the "stuck on
   360→   360→      // thinking" UI bug where the AI's response never appears during
   361→   361→      // streaming (only after a page reload fetches the persisted message).
   362→   362→      //
   363→   363→      // Fix: on the FIRST event that carries a server messageId different
   364→   364→      // from our local assistantId, rename the message in the store. After
   365→   365→      // that, all subsequent events (text.delta, tool.start, etc.) will
   366→   366→      // find the message by its new (server) ID.
   367→   367→      let idReconciled = false;
   368→   368→      // Track the CURRENT message id (starts as the client-generated id,
   369→   369→      // becomes the server's id after reconciliation).
   370→   370→      let currentMessageId = assistantId;
   371→   371→
   372→   372→      try {
   373→   373→        for await (const evt of consumeReadableStream(stream, { signal })) {
   374→   374→          if (signal.aborted || timedOut) break;
   375→   375→          resetWatchdog();
   376→   376→
   377→   377→          // Reconcile the message ID on the first event that has a messageId.
   378→   378→          if (!idReconciled && evt.messageId && evt.messageId !== assistantId) {
   379→   379→            renameMessage(assistantId, evt.messageId);
   380→   380→            currentMessageId = evt.messageId;
   381→   381→            idReconciled = true;
   382→   382→          }
   383→   383→
   384→   384→          // Track browser tool calls that need client-side execution. We
   385→   385→          // collect them from tool.start events (so we know the tool name)
   386→   386→          // AND from the explicit browser.tools_pending event (which is the
   387→   387→          // server's authoritative "these need execution now" signal).
   388→   388→          if (evt.type === "browser.tools_pending") {
   389→   389→            for (const id of evt.callIds) {
   390→   390→              if (!browserPending.includes(id)) browserPending.push(id);
   391→   391→            }
   392→   392→          }
   393→   393→
   394→   394→          // Route every event to the dispatcher → stores. IMMEDIATE, no batch.
   395→   395→          dispatcher.dispatch(evt);
   396→   396→
   397→   397→          if (evt.type === "stream.complete") break;
   398→   398→        }
   399→   399→      } catch (err) {
   400→   400→        if (signal.aborted) {
   401→   401→          // User cancelled — preserve partial content (PRD §34).
   402→   402→        } else {
   403→   403→          const msg = err instanceof Error ? err.message : "Stream error";
   404→   404→          addErrorSegment(currentMessageId, msg);
   405→   405→          toast.error("AI stream error", { description: msg });
   406→   406→        }
   407→   407→      } finally {
   408→   408→        clearWatchdog();
   409→   409→      }
   410→   410→
   411→   411→      return { browserPending };
   412→   412→    },
   413→   413→    [addErrorSegment, renameMessage]
   414→   414→  );
   415→   415→
   416→   416→  // ---------- Browser tool execution + /continue loop ----------
   417→   417→  //
   418→   418→  // After the primary stream ends, if there are pending browser tools, run
   419→   419→  // them against the preview iframe and call /continue with the results. The
   420→   420→  // /continue stream may produce MORE browser tools — we loop until none
   421→   421→  // remain (PRD §27 event ordering, fix-3 in worklog).
   422→   422→  const executeBrowserToolsAndContinue = React.useCallback(
   423→   423→    async (
   424→   424→      chatIdLocal: string,
   425→   425→      initialCallIds: string[],
   426→   426→      assistantId: string,
   427→   427→      signal: AbortSignal
   428→   428→    ) => {
   429→   429→      const b = bridgeRef.current;
   430→   430→      if (!b) {
   431→   431→        toast.error("Preview not available", {
   432→   432→          description: "Open the Preview tab to enable browser automation.",
   433→   433→        });
   434→   434→        return;
   435→   435→      }
   436→   436→
   437→   437→      let pendingCallIds = initialCallIds;
   438→   438→      let iteration = 0;
   439→   439→      const MAX_CONTINUE_ITERATIONS = 25;
   440→   440→      while (
   441→   441→        pendingCallIds.length > 0 &&
   442→   442→        !signal.aborted &&
   443→   443→        iteration < MAX_CONTINUE_ITERATIONS
   444→   444→      ) {
   445→   445→        iteration++;
   446→   446→        const results: { callId: string; result?: unknown; error?: string }[] =
   447→   447→          [];
   448→   448→        for (const callId of pendingCallIds) {
   449→   449→          const t = useToolStore.getState().tools[callId];
   450→   450→          if (!t) {
   451→   451→            results.push({ callId, error: "Tool call not found" });
   452→   452→            continue;
   453→   453→          }
   454→   454→          try {
   455→   455→            const r = await b.execute(t.tool, t.parsedArguments ?? {}, callId);
   456→   456→            toolResult(
   457→   457→              callId,
   458→   458→              r.error ? "error" : "success",
   459→   459→              r.result,
   460→   460→              r.error,
   461→   461→              t.label,
   462→   462→              t.detail
   463→   463→            );
   464→   464→            setToolResult(assistantId, callId, {
   465→   465→              status: r.error ? "error" : "success",
   466→   466→              result: r.result,
   467→   467→              error: r.error,
   468→   468→            });
   469→   469→            toolComplete(callId);
   470→   470→            results.push({ callId, result: r.result, error: r.error });
   471→   471→          } catch (err) {
   472→   472→            const errMsg =
   473→   473→              err instanceof Error ? err.message : "Execution failed";
   474→   474→            toolResult(callId, "error", undefined, errMsg, t.label, t.detail);
   475→   475→            setToolResult(assistantId, callId, {
   476→   476→              status: "error",
   477→   477→              error: errMsg,
   478→   478→            });
   479→   479→            toolComplete(callId);
   480→   480→            results.push({ callId, error: errMsg });
   481→   481→          }

   483→   483→
   484→   484→        setPendingBrowserTools([]);
   485→   485→
   486→   486→        try {
   487→   487→          const stream = await api.streamContinue(
   488→   488→            chatIdLocal,
   489→   489→            { results },
   490→   490→            signal
   491→   491→          );
   492→   492→          const { browserPending } = await consumeStream(
   493→   493→            stream,
   494→   494→            assistantId,
   495→   495→            signal
   496→   496→          );
   497→   497→          pendingCallIds = browserPending;
   498→   498→        } catch (err) {
   499→   499→          if (signal.aborted) return;
   500→   500→          const errMsg =
   501→   501→            err instanceof Error ? err.message : "Continue failed";
   502→   502→          addErrorSegment(assistantId, errMsg);
   503→   503→          toast.error("AI continue failed", { description: errMsg });
   504→   504→          return;
   505→   505→        }
   506→   506→      }
   507→   507→      if (iteration >= MAX_CONTINUE_ITERATIONS && pendingCallIds.length > 0) {
   508→   508→        addErrorSegment(
   509→   509→          assistantId,
   510→   510→          "Reached the maximum number of browser-tool continuation rounds (25). The agent may be stuck in a loop — try rephrasing your request."
   511→   511→        );
   512→   512→      }
   513→   513→    },
   514→   514→    [consumeStream, toolResult, toolComplete, setToolResult, setPendingBrowserTools, addErrorSegment]
